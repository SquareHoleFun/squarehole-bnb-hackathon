## Exact-size exports

Run this from the repository root with Node and ImageMagick installed.
It writes the exports and their manifest. It leaves every source unchanged.
The resize preserves shape, then crops equally from the center.
Transparent pixels are placed on the paper color. Metadata is stripped from exports only.

```sh
node --input-type=module <<'JS'
import { execFileSync } from 'node:child_process';
import { createHash } from 'node:crypto';
import { readFileSync, writeFileSync } from 'node:fs';
import { sizes, original, pngSize } from './examples/boxes/junkyard/check.mjs';
const kit = 'examples/boxes/junkyard/';
const hash = data => createHash('sha256').update(data).digest('hex');
const assets = {};
for (const [name, [width, height]] of Object.entries(sizes)) {
  const source = readFileSync(kit + original(name));
  const size = `${width}x${height}`;
  execFileSync('magick', [kit + original(name), '-background', '#F5EBD8',
    '-alpha', 'remove', '-alpha', 'off', '-colorspace', 'sRGB',
    '-filter', 'Lanczos', '-resize', size + '^', '-gravity', 'center',
    '-extent', size, '-depth', '8', '-strip', '-define', 'png:exclude-chunks=all',
    '-define', 'png:color-type=2', kit + name]);
  assets[name] = { size: [width, height], sha256: hash(readFileSync(kit + name)),
    originalSize: pngSize(source), originalSha256: hash(source) };
}
const firstSource = readFileSync(kit + 'art/scenes/scene-04-first-original.png');
Object.assign(assets['art/scenes/scene-04.png'], {
  earlierOriginalSize: pngSize(firstSource), earlierOriginalSha256: hash(firstSource),
});
writeFileSync(kit + 'assets.json', JSON.stringify(assets, null, 2) + '\n');
JS
```
